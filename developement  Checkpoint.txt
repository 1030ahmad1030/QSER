# QSERPINN User Experience — Complete

## Introduction

QSERPINN is the main class that ties together all QSER components: Geometry, Physics, Operators, Energy Tracking, and Validation. It provides a unified interface for solving PDEs with Physics-Informed Neural Networks (PINNs) on **any geometry without requiring a mesh**.


## Quick Start

```python
from QSER import QSERPINN
from QSER.Geometry import Geometry
from QSER.PINNs.physics import TransportPINN
import torch.nn as nn
import torch

# 1. Import geometry (no mesh!)
geo = Geometry.from_stl('room.stl')

# 2. Define physics
physics = TransportPINN(v=0.1, D=0.05, lambda_=0.01, F=1.0)

# 3. Define neural network (S, E, R)
class MyNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(3, 64), nn.Tanh(),  # (x, y, z, t) → 3 inputs
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 3)               # S, E, R
        )
    def forward(self, x, y, z, t):
        return self.net(torch.cat([x, y, z, t], dim=-1))

# 4. Create and train
pinn = QSERPINN(geo, physics, mode='qser', backend='torch')
pinn.set_model(MyNN())
pinn.sample_points(10000, 2000)
pinn.train(epochs=10000)

# 5. Predict
S, E, R = pinn.predict(new_points)

# 6. Validate
validation = pinn.validate(S, E, R)
print(f"All passed: {validation['all_passed']}")
```

### 2. Define Physics

```python
from QSER.PINNs.physics import TransportPINN

# Constant coefficients
physics = TransportPINN(
    v=0.1,
    D=0.05,
    lambda_=0.01,
    F=1.0
)

# Field coefficients
physics = TransportPINN(
    v=lambda x, t: 0.1 + 0.3 * torch.sin(2 * torch.pi * x[..., 0] / 10.0),
    D=lambda x, t: 0.02 + 0.01 * torch.exp(-(x[..., 0] - 5)**2 / 2),
    lambda_=0.01,
    F=lambda x, t: 2.0 * torch.exp(-((x[..., 0] - 5)**2 + (x[..., 1] - 5)**2) / 0.5)
)

# Custom physics
from QSER.PINNs.physics import CustomPhysics

def my_Lc(field, x, t):
    field_t = torch.autograd.grad(field, t, ...)[0]
    return field_t

def my_Ld(field, x, t):
    lap = Laplacian()
    return -lap.compute(field, x)

physics = CustomPhysics(
    Lc=my_Lc,
    Ld=my_Ld,
    source=lambda x, t: 0.0
)
```

### 3. Define Neural Network

```python
import torch.nn as nn
import torch

# Standard mode (R only)
class StandardNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(3, 64), nn.Tanh(),  # (x, y, z, t)
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 1)               # R only
        )
    def forward(self, x, y, z, t):
        return self.net(torch.cat([x, y, z, t], dim=-1))

# QSER mode (S, E, R)
class QSERNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(3, 64), nn.Tanh(),
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 3)               # S, E, R
        )
    def forward(self, x, y, z, t):
        return self.net(torch.cat([x, y, z, t], dim=-1))

# Steady-state (no time)
class SteadyNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(2, 64), nn.Tanh(),  # (x, y) only
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 64), nn.Tanh(),
            nn.Linear(64, 1)
        )
    def forward(self, x, y, t=None):
        return self.net(torch.cat([x, y], dim=-1))
```

## Updated `TransportPINN` Signature

```python
class TransportPINN(PhysicsBase):
    def __init__(
        self,
        v=0.1,
        D=0.05,
        lambda_=0.01,
        F=1.0,
        backend='torch',
        gradient_method='5point',        # ← NEW
        integration_method='trapezoidal', # ← NEW
        n_samples=10000                   # ← NEW
    ):
        super().__init__(v, D, lambda_, F, backend)
        
        # Initialize operators with gradient method
        self.grad = Gradient(
            backend=backend,
            method=gradient_method
        )
        self.lap = Laplacian(backend=backend)
        self.time_grad = TimeGradient(backend=backend)
        
        # Store integration settings
        self.integration_method = integration_method
        self.n_samples = n_samples
```
